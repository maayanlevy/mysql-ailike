cmake_minimum_required(VERSION 3.20)
project(mysql_ailike VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
find_package(CURL REQUIRED)
find_path(MYSQL_INCLUDE_DIR mysql/udf_registration_types.h PATHS /usr/include/mysql REQUIRED)
set(MYSQL_SOURCE_INCLUDE_DIR "" CACHE PATH "MySQL server source include directory")
find_library(MYSQL_SERVICES_LIBRARY mysqlservices PATHS /usr/lib64/mysql REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic)
add_library(ailike_udf MODULE src/udf.cc src/typesafe.cc)
target_include_directories(ailike_udf PRIVATE ${MYSQL_SOURCE_INCLUDE_DIR} ${MYSQL_INCLUDE_DIR})
target_link_libraries(ailike_udf PRIVATE CURL::libcurl ${MYSQL_SERVICES_LIBRARY})
set_target_properties(ailike_udf PROPERTIES PREFIX "")
add_library(ailike_rewrite MODULE src/plugin.cc src/rewrite.cc)
target_include_directories(ailike_rewrite PRIVATE ${MYSQL_SOURCE_INCLUDE_DIR} ${MYSQL_INCLUDE_DIR})
target_compile_definitions(ailike_rewrite PRIVATE MYSQL_DYNAMIC_PLUGIN)
target_link_libraries(ailike_rewrite PRIVATE ${MYSQL_SERVICES_LIBRARY})
set_target_properties(ailike_rewrite PROPERTIES PREFIX "")
if(BUILD_TESTING)
  add_executable(rewrite_test tests/rewrite_test.cc src/rewrite.cc)
  target_include_directories(rewrite_test PRIVATE src)
  add_test(NAME rewrite COMMAND rewrite_test)
  add_executable(typesafe_test tests/typesafe_test.cc src/typesafe.cc)
  target_include_directories(typesafe_test PRIVATE src)
  target_link_libraries(typesafe_test PRIVATE CURL::libcurl)
  add_test(NAME typesafe COMMAND typesafe_test)
endif()
install(TARGETS ailike_udf ailike_rewrite LIBRARY DESTINATION lib/mysql/plugin)
